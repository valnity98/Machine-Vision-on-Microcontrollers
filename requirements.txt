# STM32 Edge Vision — unified requirements
#
# Covers both the PC-side Reference Dashboard (PySide6 GUI) and the
# scripts in "ML training/src" (training, dataset check, TFLite verification).
#
# Python 3.10 to 3.12 (TensorFlow 2.16.2 has no wheels for Python 3.13+).
# Without TensorFlow, the dashboard and the PC-side inference script also run on
# Python 3.14 (checked with PySide6 6.12, Pillow, numpy 2.x and ai-edge-litert 2.3.0).
#
# The ST Model Zoo training workflow has its own requirements file and needs
# its own environment; see docs/training.md.
#
# Quick start:
#   python -m venv .venv
#   .venv\Scripts\activate        # Windows
#   source .venv/bin/activate     # Linux / macOS
#   pip install -r requirements.txt

# ── GUI (Reference Dashboard) ────────────────────────────────────────────────
PySide6>=6.6.0          # Qt widgets + QtSerialPort (included)

# ── Image handling ───────────────────────────────────────────────────────────
Pillow>=10.3.0

# ── Deep Learning (training + TFLite inference) ──────────────────────────────
tensorflow==2.16.2      # training, TFLite converter, PTQ quantisation
# Lighter alternative for inference only (no training):
# ai-edge-litert>=1.0.1

# ── Numerics ─────────────────────────────────────────────────────────────────
numpy>=1.24.0,<2.0      # TF 2.16 is not compatible with numpy 2.x
